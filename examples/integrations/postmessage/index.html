Send files to clinker

This external app demo loads two bundled GenBank examples into browser File objects, opens clinker, waits for it to signal readiness, and hands the files over with postMessage.

Clinker stages the files for review. It does not start analysis automatically.

Minimal code for an external app 
function sendFilesToClinker(files, clinkerUrl) {
  const target = new URL(clinkerUrl);
  target.searchParams.set("integration", "postmessage");
  const requestId = crypto.randomUUID();

  return new Promise((resolve, reject) => {
    let clinker;
    const timeout = setTimeout(() => finish(new Error("Clinker did not respond")), 30000);
    function finish(error, result) {
      clearTimeout(timeout);
      window.removeEventListener("message", onMessage);
      if (error) reject(error);
      else resolve(result);
    }
    function onMessage(event) {
      if (event.source !== clinker || event.origin !== target.origin) return;
      if (event.data?.type === "clinker:ready") {
        clinker.postMessage({ type: "clinker:import-files", requestId, files }, target.origin);
      } else if (event.data?.type === "clinker:files-received"
          && event.data.requestId === requestId) {
        if (event.data.ok) finish(null, clinker);
        else finish(new Error(event.data.error));
      }
    }
    window.addEventListener("message", onMessage);
    clinker = window.open(target.href, "clinker-import");
    if (!clinker) finish(new Error("Allow the popup, then try again"));
  });
}

// `files` can be generated directly by the external app:
const files = [new File([genbankText], "cluster.gbk", { type: "text/plain" })];
sendFilesToClinker(files, "https://your-clinker.pages.dev/");
